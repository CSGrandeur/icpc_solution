### `Calendar Cubes`

> difficulty: 3

[QOJ 17761](https://qoj.ac/contest/3588/problem/17761)

分类讨论较繁琐；根据数据范围，暴力枚举虽在时限内不可接受，但本地运行可承受（远小于 $9^{12}$），本地 DFS 打表后嵌入提交即可。

int cs[12];
int res[101][12];
void DFS(int cur) {
    if(cur == 12) {
        bool vis[100] = {0};
        for(int i = 0; i < 6; i ++) {
            for(int j = 6; j < 12; j ++) {
                vis[cs[i] * 10 + cs[j]] = true;
                vis[cs[i] + cs[j] * 10] = true;
                if(cs[i] == 6) {
                    vis[90 + cs[j]] = true;
                    vis[9 + cs[j] * 10] = true;
                }
                if(cs[j] == 6) {
                    vis[cs[i] * 10 + 9] = true;
                    vis[cs[i] + 90] = true;
                }
            }
        }
        int nt = 1;
        for(; nt <= 99; nt ++) {
            if(!vis[nt]) {
                break;
            }
        }
        if(res[nt][0] == -1) {
            memcpy(res[nt], cs, sizeof(cs));
        }
        return;
    }
    for(int i = (cur == 0 || cur == 6) ? 0 : cs[cur - 1]; i <= 8; i ++) {
        cs[cur] = i;
        DFS(cur + 1);
    }
}

        printf("}");
        printf(",\n" + (i == 99));
    }
    printf("};\n");
}
